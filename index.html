from flask import Flask, render_template_string import requests app = Flask(__name__) HTML_TEMPLATE = """ Crypto Dashboard

Crypto Dashboard

Top Coins

{% for coin in coins %}
{{ coin.name }} - ${{ coin.current_price }}
{% endfor %}
""" @app.route("/") def home(): url = "https://api.coingecko.com/api/v3/coins/markets" params = { "vs_currency": "usd", "order": "market_cap_desc", "per_page": 15, "page": 1 } coins = requests.get(url, params=params).json() return render_template_string(HTML_TEMPLATE, coins=coins) if __name__ == "__main__": app.run(debug=True)
